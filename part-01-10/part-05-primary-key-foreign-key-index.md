## 1. Primary Key ในเชิงลึก

### 1.1 Primary Key คืออะไร?

Primary Key คือ constraint ที่:
1. **Uniquely identifies** แต่ละ row ในตาราง
2. **ไม่สามารถเป็น NULL** ได้
3. **Automatically creates an index** (ใน PostgreSQL สร้าง B-tree index)
4. **Enforces entity integrity** — ทุก row ต้องมี identity

```sql
-- สร้าง Primary Key แบบต่างๆ
-- วิธีที่ 1: Inline definition
CREATE TABLE pk_inline (
    id      SERIAL PRIMARY KEY,
    name    VARCHAR(100)
);

-- วิธีที่ 2: Table constraint
CREATE TABLE pk_table_constraint (
    id      SERIAL,
    name    VARCHAR(100),
    
    CONSTRAINT pk_id PRIMARY KEY (id)
);

-- วิธีที่ 3: Composite Primary Key
CREATE TABLE pk_composite (
    user_id     INTEGER,
    product_id  INTEGER,
    quantity    INTEGER,
    
    PRIMARY KEY (user_id, product_id)
);

-- เพิ่ม Primary Key หลังจากสร้าง table
CREATE TABLE pk_add_later (
    id      INTEGER,
    name    VARCHAR(100)
);

ALTER TABLE pk_add_later ADD PRIMARY KEY (id);

-- ลบ Primary Key
ALTER TABLE pk_add_later DROP CONSTRAINT pk_add_later_pkey;

-- ดู Primary Key constraint
SELECT 
    tc.constraint_name,
    tc.table_name,
    kcu.column_name
FROM information_schema.table_constraints tc
JOIN information_schema.key_column_usage kcu
    ON tc.constraint_name = kcu.constraint_name
WHERE tc.constraint_type = 'PRIMARY KEY'
  AND tc.table_schema = 'public';
```

### 1.2 Properties ของ Primary Key

```sql
-- Test 1: PRIMARY KEY ไม่ยอมรับ NULL
INSERT INTO pk_inline (id, name) VALUES (NULL, 'Test');  -- ERROR!

-- Test 2: PRIMARY KEY ไม่ยอมรับ duplicate
INSERT INTO pk_inline (name) VALUES ('Alice');  -- id = 1
INSERT INTO pk_inline (name) VALUES ('Bob');    -- id = 2
INSERT INTO pk_inline (id, name) VALUES (1, 'Charlie');  -- ERROR: duplicate key!

-- Test 3: Primary Key Index สร้างอัตโนมัติ
\d pk_inline  -- จะเห็น index "pk_inline_pkey" btree (id)

-- ดู primary key ทำให้ query เร็วขึ้นอย่างไร
EXPLAIN SELECT * FROM pk_inline WHERE id = 1;
-- Output: Index Scan using pk_inline_pkey on pk_inline  (เร็วมาก!)

EXPLAIN SELECT * FROM pk_inline WHERE name = 'Alice';
-- Output: Seq Scan on pk_inline  (ช้า — ไม่มี index บน name)
```

### 1.3 SERIAL vs IDENTITY

```sql
-- SERIAL: เก่ากว่า (syntax PostgreSQL-specific)
CREATE TABLE serial_demo (
    id SERIAL PRIMARY KEY
);
-- เทียบเท่ากับ:
CREATE SEQUENCE serial_demo_id_seq;
CREATE TABLE serial_demo_manual (
    id INTEGER DEFAULT nextval('serial_demo_id_seq') NOT NULL
);

-- GENERATED AS IDENTITY: SQL standard (PostgreSQL 10+)
CREATE TABLE identity_demo (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    -- หรือ
    id2 INTEGER GENERATED BY DEFAULT AS IDENTITY,  -- อนุญาต override
    name VARCHAR(100)
);

-- ความแตกต่าง ALWAYS vs BY DEFAULT:
-- GENERATED ALWAYS: ไม่สามารถ insert ค่า id เองได้ (ยกเว้น OVERRIDING SYSTEM VALUE)
INSERT INTO identity_demo (name) VALUES ('test');  -- OK, id auto-generated
-- INSERT INTO identity_demo (id, name) VALUES (100, 'test');  -- ERROR with ALWAYS

-- GENERATED BY DEFAULT: อนุญาตให้ specify id เองได้
INSERT INTO identity_demo (id2, name) VALUES (100, 'test');  -- OK

-- Sequence operations สำหรับ IDENTITY
ALTER TABLE identity_demo ALTER COLUMN id RESTART WITH 1000;
ALTER TABLE identity_demo ALTER COLUMN id SET INCREMENT BY 10;
```

## 2. เปรียบเทียบ ID Strategies

### 2.1 Sequential Integer (SERIAL/BIGSERIAL)

```sql
-- Pros and Cons
-- ✅ เล็กที่สุด (4 หรือ 8 bytes)
-- ✅ เรียงตามลำดับ → B-tree index fragmentation ต่ำ
-- ✅ อ่านง่าย
-- ✅ JOIN เร็วที่สุด (int comparison)
-- ❌ Sequential ทำให้เดา URL ง่าย (/users/1, /users/2)
-- ❌ ต้องใช้ DB เพื่อสร้าง ID (ไม่สามารถสร้างบน client)
-- ❌ Multi-DB merge ยาก

CREATE TABLE users_serial (
    user_id BIGSERIAL PRIMARY KEY,
    email   VARCHAR(255) UNIQUE
);

-- ทดสอบ insert performance
INSERT INTO users_serial (email)
SELECT 'user' || i || '@example.com'
FROM generate_series(1, 100000) AS i;

\timing on
SELECT * FROM users_serial WHERE user_id = 50000;
-- Index Scan: ~0.1ms
```

### 2.2 UUID v4 (Random)

```sql
-- ✅ Globally unique
-- ✅ ปลอดภัย (ไม่สามารถเดาได้)
-- ✅ สร้างบน client ได้
-- ✅ Multi-DB merge ง่าย
-- ❌ ใหญ่กว่า 4x (16 bytes vs 4 bytes)
-- ❌ Random → B-tree fragmentation สูง → write ช้ากว่า
-- ❌ อ่านยาก

CREATE TABLE users_uuid (
    user_id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    email   VARCHAR(255) UNIQUE
);

-- ทดสอบ insert (จะช้ากว่า serial เพราะ index fragmentation)
INSERT INTO users_uuid (email)
SELECT 'user' || i || '@example.com'
FROM generate_series(1, 100000) AS i;

-- เปรียบเทียบ index size
SELECT 
    relname,
    pg_size_pretty(pg_total_relation_size(oid)) AS total_size
FROM pg_class
WHERE relname IN ('users_serial', 'users_uuid');
-- users_uuid จะใหญ่กว่า users_serial ประมาณ 2-3x
```

### 2.3 UUID v7 (Time-ordered)

```sql
-- UUID v7: เรียงตาม timestamp แต่ยังคงเป็น UUID
-- ✅ Globally unique
-- ✅ Time-ordered (ลด B-tree fragmentation)
-- ✅ Extract timestamp ได้
-- ❌ ยังคงใหญ่กว่า integer (16 bytes)

-- PostgreSQL 17 มี gen_random_uuid() แต่ยังไม่มี UUIDv7 built-in
-- สร้างด้วย extension หรือ custom function:

CREATE OR REPLACE FUNCTION uuid_generate_v7() RETURNS UUID AS $$
DECLARE
    unix_ts_ms BYTEA;
    uuid_bytes BYTEA;
BEGIN
    unix_ts_ms = DECODE(LPAD(TO_HEX((EXTRACT(EPOCH FROM CLOCK_TIMESTAMP()) * 1000)::BIGINT), 12, '0'), 'hex');
    uuid_bytes = unix_ts_ms || gen_random_bytes(10);
    
    -- Set version (7) and variant bits
    uuid_bytes = SET_BYTE(uuid_bytes, 6, (GET_BYTE(uuid_bytes, 6) & 15) | 112);  -- version 7
    uuid_bytes = SET_BYTE(uuid_bytes, 8, (GET_BYTE(uuid_bytes, 8) & 63) | 128);  -- variant
    
    RETURN ENCODE(uuid_bytes, 'hex')::UUID;
END;
$$ LANGUAGE plpgsql VOLATILE;

CREATE TABLE users_uuid7 (
    user_id UUID DEFAULT uuid_generate_v7() PRIMARY KEY,
    email   VARCHAR(255) UNIQUE
);

-- UUIDv7 จะเรียงตามเวลา
SELECT uuid_generate_v7();
SELECT uuid_generate_v7();  -- ค่าที่ 2 จะมากกว่าค่าที่ 1 เสมอ
```

## 3. Foreign Key ในเชิงลึก

### 3.1 Referential Integrity

```sql
-- Foreign Key บังคับ Referential Integrity:
-- ค่าใน FK column ต้องมีอยู่ใน Referenced column

CREATE TABLE fk_categories (
    category_id SERIAL PRIMARY KEY,
    name        VARCHAR(100)
);

CREATE TABLE fk_products (
    product_id  SERIAL PRIMARY KEY,
    category_id INTEGER REFERENCES fk_categories(category_id),
    name        VARCHAR(255)
);

-- ทดสอบ Referential Integrity
INSERT INTO fk_categories VALUES (1, 'Electronics');
INSERT INTO fk_products VALUES (1, 1, 'iPhone');      -- OK: category 1 exists

-- ERROR: foreign key violation
-- INSERT INTO fk_products VALUES (2, 999, 'Laptop');  -- category 999 ไม่มี!

-- ลบ parent ที่มี children
-- DELETE FROM fk_categories WHERE category_id = 1;  -- ERROR: ยังมี products อ้างอิงอยู่
```

### 3.2 ON DELETE Options

```sql
-- 1. RESTRICT (Default): ป้องกันการลบ parent ที่มี children
CREATE TABLE fk_restrict (
    id          SERIAL PRIMARY KEY,
    category_id INTEGER REFERENCES fk_categories(category_id) ON DELETE RESTRICT
);

-- 2. CASCADE: ลบ children อัตโนมัติเมื่อลบ parent
CREATE TABLE fk_cascade (
    id          SERIAL PRIMARY KEY,
    category_id INTEGER REFERENCES fk_categories(category_id) ON DELETE CASCADE
);

-- ลบ category → ลบ products ใน cascade table ด้วย
DELETE FROM fk_categories WHERE category_id = 1;  
-- ถ้า fk_cascade มี rows อ้างอิง category 1 → ถูกลบไปด้วย

-- 3. SET NULL: ตั้ง FK เป็น NULL
CREATE TABLE fk_set_null (
    id          SERIAL PRIMARY KEY,
    category_id INTEGER REFERENCES fk_categories(category_id) ON DELETE SET NULL
);

-- 4. SET DEFAULT: ตั้ง FK เป็น default value
CREATE TABLE fk_set_default (
    id          SERIAL PRIMARY KEY,
    category_id INTEGER DEFAULT 1 REFERENCES fk_categories(category_id) ON DELETE SET DEFAULT
);

-- 5. NO ACTION: เหมือน RESTRICT แต่ check ตอนท้าย transaction
-- (ใช้กับ DEFERRABLE INITIALLY DEFERRED)
CREATE TABLE fk_no_action (
    id          SERIAL PRIMARY KEY,
    category_id INTEGER REFERENCES fk_categories(category_id) ON DELETE NO ACTION
);
```

### 3.3 ON UPDATE Options (เหมือน ON DELETE)

```sql
-- CASCADE: อัปเดต FK value เมื่อ PK เปลี่ยน
CREATE TABLE fk_update_cascade (
    id          SERIAL PRIMARY KEY,
    category_id INTEGER REFERENCES fk_categories(category_id) 
        ON DELETE CASCADE 
        ON UPDATE CASCADE
);

-- อัปเดต category_id ใน parent
-- UPDATE fk_categories SET category_id = 100 WHERE category_id = 1;
-- → FK ใน fk_update_cascade จะอัปเดตจาก 1 เป็น 100 อัตโนมัติ
-- (แต่ตามปกติเราไม่ควร update PK)
```

### 3.4 Performance Impact ของ Foreign Keys

```sql
-- FK มี performance overhead เพราะ:
-- 1. ทุก INSERT/UPDATE บน child table → ต้อง check parent exists
-- 2. ทุก DELETE/UPDATE บน parent table → ต้อง check no children

-- เพื่อ performance: สร้าง index บน FK column!
-- PostgreSQL ไม่สร้าง index บน FK โดยอัตโนมัติ (ต่างจาก PK)

-- ตรวจหา FK ที่ไม่มี index (อาจทำให้ delete parent ช้า)
SELECT
    tc.table_name,
    kcu.column_name,
    ccu.table_name AS foreign_table
FROM information_schema.table_constraints AS tc
JOIN information_schema.key_column_usage AS kcu
    ON tc.constraint_name = kcu.constraint_name
JOIN information_schema.constraint_column_usage AS ccu
    ON ccu.constraint_name = tc.constraint_name
WHERE tc.constraint_type = 'FOREIGN KEY'
  AND NOT EXISTS (
      SELECT 1 FROM pg_indexes
      WHERE tablename = tc.table_name
        AND indexdef LIKE '%(' || kcu.column_name || '%'
  );

-- สร้าง index บน FK columns
CREATE INDEX idx_fk_products_category ON fk_products(category_id);
CREATE INDEX idx_fk_orders_customer ON shop.orders(customer_id);
CREATE INDEX idx_fk_order_items_order ON shop.order_items(order_id);
CREATE INDEX idx_fk_order_items_product ON shop.order_items(product_id);
```

### 3.5 Deferrable Foreign Keys

```sql
-- ปกติ: FK checked immediately after each statement
-- DEFERRABLE: FK checked at END of transaction

-- ใช้ประโยชน์เมื่อต้องการ insert circular references:
CREATE TABLE fk_deferred_a (
    id      INTEGER PRIMARY KEY,
    b_id    INTEGER
);

CREATE TABLE fk_deferred_b (
    id      INTEGER PRIMARY KEY,
    a_id    INTEGER REFERENCES fk_deferred_a(id) DEFERRABLE INITIALLY DEFERRED
);

ALTER TABLE fk_deferred_a
ADD CONSTRAINT fk_a_b FOREIGN KEY (b_id) 
REFERENCES fk_deferred_b(id) DEFERRABLE INITIALLY DEFERRED;

-- Insert circular references ใน transaction เดียว
BEGIN;
INSERT INTO fk_deferred_a VALUES (1, 2);  -- อ้างอิง b_id=2 ที่ยังไม่มี
INSERT INTO fk_deferred_b VALUES (2, 1);  -- อ้างอิง a_id=1 ที่สร้างไปแล้ว
COMMIT;  -- ตรวจสอบ FK ตอนนี้ → ทั้งคู่ valid
```

## 4. Index Types ใน PostgreSQL

### 4.1 B-Tree Index (Default)

```sql
-- B-Tree: Balanced Binary Tree
-- ใช้กับ: <, <=, =, >=, >, BETWEEN, IN, LIKE 'prefix%', IS NULL/NOT NULL
-- ดีสำหรับ: เกือบทุก use case ทั่วไป

-- สร้าง B-Tree index
CREATE INDEX idx_btree_price ON shop.products(price);
CREATE INDEX idx_btree_status ON shop.orders(status);
CREATE INDEX idx_btree_email ON shop.customers(email);

-- สิ่งที่ B-Tree ทำได้:
EXPLAIN SELECT * FROM shop.products WHERE price = 42900;    -- Index Scan
EXPLAIN SELECT * FROM shop.products WHERE price > 10000;   -- Index Scan
EXPLAIN SELECT * FROM shop.products WHERE price BETWEEN 1000 AND 5000;  -- Index Scan
EXPLAIN SELECT * FROM shop.customers WHERE email LIKE 'alice%';  -- Index Scan (prefix)

-- สิ่งที่ B-Tree ทำไม่ได้:
EXPLAIN SELECT * FROM shop.customers WHERE email LIKE '%alice%';  -- Seq Scan (% ข้างหน้า)
EXPLAIN SELECT * FROM shop.customers WHERE LOWER(email) = 'alice@example.com';  -- Seq Scan (function)
```

### 4.2 Hash Index

```sql
-- Hash Index: ใช้ hash function
-- ใช้กับ: = เท่านั้น (ไม่รองรับ range)
-- ข้อดี: เร็วกว่า B-Tree สำหรับ equality query เล็กน้อย
-- ข้อเสีย: ไม่รองรับ range, ordering

CREATE INDEX idx_hash_status ON shop.orders USING HASH(status);

-- ใช้ได้: equality เท่านั้น
EXPLAIN SELECT * FROM shop.orders WHERE status = 'completed';  -- Hash Index

-- ไม่สามารถใช้สำหรับ range
EXPLAIN SELECT * FROM shop.orders WHERE status > 'completed';  -- Seq Scan
```

### 4.3 GIN Index (Generalized Inverted Index)

```sql
-- GIN: สำหรับ multi-valued data
-- ใช้กับ: Arrays, JSONB, full-text search, tsvector

-- Array GIN index
CREATE INDEX idx_gin_tags ON blog_posts USING GIN(tags);
EXPLAIN SELECT * FROM blog_posts WHERE tags @> ARRAY['postgresql'];

-- JSONB GIN index
CREATE INDEX idx_gin_attributes ON products_flexible USING GIN(attributes);
EXPLAIN SELECT * FROM products_flexible WHERE attributes @> '{"color": "white"}';

-- Full-text search GIN index
CREATE INDEX idx_gin_fts ON shop.products 
    USING GIN(to_tsvector('english', name || ' ' || COALESCE(description, '')));

EXPLAIN SELECT * FROM shop.products 
WHERE to_tsvector('english', name || ' ' || COALESCE(description, ''))
    @@ to_tsquery('english', 'iphone & pro');

-- GIN vs GiST สำหรับ full-text:
-- GIN: เร็วกว่าสำหรับ query, ช้ากว่าสำหรับ update
-- GiST: ช้ากว่าสำหรับ query, เร็วกว่าสำหรับ update
```

### 4.4 GiST Index (Generalized Search Tree)

```sql
-- GiST: สำหรับ geometric data, ranges, full-text
-- ใช้กับ: PostGIS geometries, Range types, full-text (tsvector)

-- Range type GiST index
CREATE INDEX idx_gist_dates ON type_demo_ranges USING GIST(valid_dates);
EXPLAIN SELECT * FROM type_demo_ranges WHERE valid_dates @> '2024-06-15'::DATE;

-- Exclusion constraint (ใช้ GiST automatically)
-- ดูตัวอย่างใน hotel_bookings จากบทที่ 4

-- PostGIS (geographic data)
-- CREATE INDEX idx_gist_location ON locations USING GIST(geom);
-- SELECT * FROM locations WHERE ST_DWithin(geom, ST_Point(100.5, 13.7), 1000);

-- Tsvector index
CREATE INDEX idx_gist_fts ON shop.products
    USING GIST(to_tsvector('english', name));
```

### 4.5 BRIN Index (Block Range Index)

```sql
-- BRIN: Block Range Index
-- ใช้กับ: columns ที่มีความสัมพันธ์กับ physical storage order
-- ข้อดี: ขนาดเล็กมาก (เล็กกว่า B-Tree 1000x+)
-- ข้อเสีย: ไม่แม่นยำ → ต้อง scan blocks หลายๆ อัน
-- เหมาะสำหรับ: time-series data, log tables, append-only tables

-- สร้าง BRIN index บน timestamp column
CREATE INDEX idx_brin_created ON shop.orders USING BRIN(created_at);

-- BRIN ทำงานได้ดีเมื่อ:
-- ข้อมูลใหม่ถูก insert ตาม order (เช่น created_at ล่าสุดอยู่ท้าย)
-- Range queries บน timestamp

EXPLAIN SELECT * FROM shop.orders WHERE created_at > '2024-06-01';
-- อาจใช้ BRIN index (ถ้าตาราง append-only)

-- เปรียบเทียบขนาด
CREATE INDEX idx_btree_created ON shop.orders(created_at);

SELECT 
    indexname,
    pg_size_pretty(pg_relation_size(indexrelid)) AS index_size
FROM pg_stat_user_indexes
WHERE tablename = 'orders'
  AND indexname IN ('idx_btree_created', 'idx_brin_created');
-- idx_brin_created จะเล็กกว่ามาก
```

### 4.6 SP-GiST Index (Space-Partitioned GiST)

```sql
-- SP-GiST: สำหรับ non-balanced data structures
-- ใช้กับ: text (prefix), ranges, points

-- ตัวอย่าง: Text prefix search
CREATE INDEX idx_spgist_name ON shop.products USING SPGIST(name);
EXPLAIN SELECT * FROM shop.products WHERE name LIKE 'iPhone%';  -- Can use SP-GiST

-- IP subnet queries
CREATE INDEX idx_spgist_ip ON audit_log USING SPGIST(client_ip inet_ops);
EXPLAIN SELECT * FROM audit_log WHERE client_ip << '192.168.0.0/16';
```

```
│ Hash      │ = only                             │ equality อย่างเดียว    │
│ GIN       │ Arrays, JSONB, full-text           │ multi-value columns    │
│ GiST      │ Geometric, Range, full-text        │ complex data types     │
│ BRIN      │ Sequential data (time-series)      │ large append-only      │
│ SP-GiST   │ Text prefix, IP ranges             │ non-balanced tree data │
└───────────┴────────────────────────────────────┴────────────────────────┘
```

## 5. เมื่อไหร่ควร/ไม่ควรสร้าง Index

### 5.1 ควรสร้าง Index เมื่อ:

```sql
-- ✅ 1. Column ใน WHERE clause ที่ใช้บ่อย
CREATE INDEX idx_orders_status ON shop.orders(status);
-- Query: SELECT * FROM orders WHERE status = 'pending'

-- ✅ 2. Column ใน JOIN ON clause
CREATE INDEX idx_order_items_order_id ON shop.order_items(order_id);
-- Query: SELECT * FROM orders o JOIN order_items oi ON o.order_id = oi.order_id

-- ✅ 3. Column ใน ORDER BY (สำหรับ pagination)
CREATE INDEX idx_orders_created_desc ON shop.orders(created_at DESC);
-- Query: SELECT * FROM orders ORDER BY created_at DESC LIMIT 10

-- ✅ 4. FK columns (performance สำคัญ)
CREATE INDEX idx_products_category_fk ON shop.products(category_id);

-- ✅ 5. Columns ที่ใช้ aggregate บ่อยๆ
CREATE INDEX idx_orders_customer_status ON shop.orders(customer_id, status);
-- Query: SELECT customer_id, COUNT(*) FROM orders WHERE status = 'completed' GROUP BY customer_id

-- ✅ 6. Selective columns (high cardinality)
-- email มี cardinality สูง (เกือบทุก row ต่างกัน) → index มีประโยชน์มาก
CREATE INDEX idx_customers_email ON shop.customers(email);

-- ✅ 7. Columns ที่ใช้ใน subqueries บ่อยๆ
CREATE INDEX idx_reviews_product ON shop.reviews(product_id);
```

### 5.2 ไม่ควรสร้าง Index เมื่อ:

```sql
-- ❌ 1. Small tables (< 1000 rows) — Seq Scan เร็วกว่า
-- PostgreSQL optimizer มักเลือก Seq Scan สำหรับ table เล็ก

-- ❌ 2. Low-selectivity columns (low cardinality)
-- boolean column มีแค่ 2 values → ถ้า 50% เป็น TRUE, index ไม่มีประโยชน์
-- (ยกเว้นใช้ Partial Index!)

-- ❌ 3. Columns ที่ update บ่อยมาก
-- ทุกครั้งที่ update column → index ต้อง update ด้วย → overhead

-- ❌ 4. Write-heavy tables
-- Insert/Update/Delete ทุกครั้งต้อง maintain indexes ทั้งหมด
-- ถ้า table มี index 10 ตัว ทุก Insert ต้อง update 10 indexes!

-- ❌ 5. Columns ที่ไม่ได้ใช้ใน WHERE/JOIN/ORDER BY
-- Dead indexes กิน disk space และ slow down writes

-- ตรวจหา unused indexes
SELECT 
    schemaname,
    tablename,
    indexname,
    idx_scan,
    pg_size_pretty(pg_relation_size(indexrelid)) AS index_size
FROM pg_stat_user_indexes
WHERE idx_scan = 0  -- ไม่เคยถูกใช้เลย!
  AND indexrelid NOT IN (
      SELECT conindid FROM pg_constraint WHERE contype IN ('p', 'u')
  )
ORDER BY pg_relation_size(indexrelid) DESC;
```

## 6. Partial Index

```sql
-- Partial Index: Index เฉพาะ subset ของ rows
-- ข้อดี: ขนาดเล็กกว่า, เร็วกว่า, เหมาะกับ low-cardinality columns

-- ตัวอย่าง 1: Index เฉพาะ active products
CREATE INDEX idx_partial_active_products 
ON shop.products(name, price)
WHERE is_active = TRUE;

-- เปรียบเทียบขนาด
CREATE INDEX idx_full_all_products ON shop.products(name, price);

SELECT 
    indexname,
    pg_size_pretty(pg_relation_size(indexrelid)) AS size
FROM pg_stat_user_indexes
WHERE tablename = 'products'
  AND indexname IN ('idx_partial_active_products', 'idx_full_all_products');

-- ตัวอย่าง 2: Index เฉพาะ pending orders (queue processing)
CREATE INDEX idx_partial_pending_orders 
ON shop.orders(created_at)
WHERE status = 'pending';

-- Query นี้จะใช้ partial index:
SELECT * FROM shop.orders 
WHERE status = 'pending' 
ORDER BY created_at;

-- Query นี้จะ Seq Scan (ไม่ match WHERE):
SELECT * FROM shop.orders WHERE status = 'completed';

-- ตัวอย่าง 3: Unique constraint เฉพาะ non-deleted records
ALTER TABLE shop.products ADD COLUMN deleted_at TIMESTAMPTZ;

CREATE UNIQUE INDEX idx_unique_active_slug 
ON shop.products(slug)
WHERE deleted_at IS NULL;

-- อนุญาตให้มี slug เดียวกันถ้า deleted
INSERT INTO shop.products (slug, name, price) VALUES ('test-product', 'Test 1', 100);
UPDATE shop.products SET deleted_at = NOW() WHERE slug = 'test-product';
INSERT INTO shop.products (slug, name, price) VALUES ('test-product', 'Test 2', 200);  -- OK!

-- ตัวอย่าง 4: Index สำหรับ High-value orders เท่านั้น
CREATE INDEX idx_partial_high_value 
ON shop.orders(customer_id, created_at)
WHERE total_amount > 50000;

-- ตัวอย่าง 5: Index เฉพาะ NULL values
CREATE INDEX idx_partial_phone_null 
ON shop.customers(customer_id)
WHERE phone IS NULL;

-- ใช้หา customers ที่ไม่มีเบอร์โทร:
SELECT * FROM shop.customers WHERE phone IS NULL;
```

## 7. Expression Index

```sql
-- Expression Index (Functional Index): Index บน expression/function result
-- ใช้เมื่อ query ใช้ function บน column

-- ตัวอย่าง 1: Case-insensitive email search
CREATE INDEX idx_expr_email_lower ON shop.customers(LOWER(email));

-- Query จะใช้ index:
SELECT * FROM shop.customers WHERE LOWER(email) = 'alice@example.com';

-- Query จะไม่ใช้ index:
SELECT * FROM shop.customers WHERE email = 'alice@example.com';  -- case-sensitive
-- → ต้อง query แบบ LOWER() จึงจะใช้ index

-- ตัวอย่าง 2: Date-only search (ตัดเวลาออก)
CREATE INDEX idx_expr_order_date ON shop.orders((created_at::DATE));

SELECT * FROM shop.orders WHERE created_at::DATE = '2024-01-15';
-- หรือ:
SELECT * FROM shop.orders WHERE DATE(created_at) = '2024-01-15';

-- ตัวอย่าง 3: Extract month สำหรับ birthday queries
CREATE INDEX idx_expr_birth_month ON shop.customers(EXTRACT(MONTH FROM birth_date)::INT);

SELECT * FROM shop.customers 
WHERE EXTRACT(MONTH FROM birth_date) = EXTRACT(MONTH FROM NOW());

-- ตัวอย่าง 4: JSON field index
CREATE INDEX idx_expr_product_color ON products_flexible((attributes->>'color'));

SELECT * FROM products_flexible WHERE attributes->>'color' = 'white';

-- ตัวอย่าง 5: Computed slug
CREATE INDEX idx_expr_name_slug ON shop.products(LOWER(REPLACE(name, ' ', '-')));

SELECT * FROM shop.products 
WHERE LOWER(REPLACE(name, ' ', '-')) = 'iphone-15-pro';

-- Important: Expression ใน CREATE INDEX ต้องตรงกับ Query expression!
-- PostgreSQL match expressions แบบ exact text match

-- ตรวจสอบว่า query ใช้ expression index:
EXPLAIN ANALYZE SELECT * FROM shop.customers WHERE LOWER(email) = 'alice@example.com';
```

## 8. Covering Index (INCLUDE)

```sql
-- Covering Index: Index ที่รวม columns พิเศษ (ไม่ใช้ indexing แต่ใช้ lookup)
-- PostgreSQL 11+ รองรับ INCLUDE clause
-- ข้อดี: Query สามารถ "Index-Only Scan" โดยไม่ต้องเข้าถึง heap

-- ปัญหา: ถ้า query ต้องการ columns ที่ไม่มีใน index → ต้องไป fetch heap
SELECT customer_id, status, total_amount  -- total_amount ไม่มีใน index
FROM shop.orders
WHERE status = 'completed';

-- EXPLAIN → Index Scan (ต้องเข้า heap สำหรับ total_amount)
EXPLAIN SELECT customer_id, status, total_amount 
FROM shop.orders WHERE status = 'completed';

-- แก้ด้วย Covering Index:
CREATE INDEX idx_covering_orders 
ON shop.orders(status) 
INCLUDE (customer_id, total_amount, created_at);

-- ตอนนี้ query สามารถ Index-Only Scan:
EXPLAIN SELECT customer_id, status, total_amount 
FROM shop.orders WHERE status = 'completed';
-- Output: Index Only Scan using idx_covering_orders

-- ตัวอย่างจริง: Product listing page
CREATE INDEX idx_covering_product_listing
ON shop.products(category_id, is_active)
INCLUDE (product_id, name, price, stock_qty);

-- Query สำหรับ listing page จะ Index-Only Scan:
SELECT product_id, name, price, stock_qty
FROM shop.products
WHERE category_id = 1 AND is_active = TRUE;

-- ข้อสำคัญ:
-- INCLUDE columns ไม่ได้ใช้ใน ORDER BY หรือ WHERE predicates
-- ใช้สำหรับ "carry along" data เท่านั้น
-- INCLUDE columns ไม่ต้องมีค่า unique ต่างจาก indexed columns
```

## 9. Multi-column Index

```sql
-- Multi-column Index: Index หลาย columns พร้อมกัน
-- Column order สำคัญมาก!

-- สร้าง composite index
CREATE INDEX idx_multi_status_created ON shop.orders(status, created_at);

-- ใช้ index ได้:
SELECT * FROM shop.orders WHERE status = 'completed';                     -- ✅ leftmost prefix
SELECT * FROM shop.orders WHERE status = 'completed' AND created_at > '2024-01-01';  -- ✅ both
SELECT * FROM shop.orders ORDER BY status, created_at;                    -- ✅ ORDER BY

-- ใช้ index ไม่ได้:
SELECT * FROM shop.orders WHERE created_at > '2024-01-01';               -- ❌ skips first column
-- (PostgreSQL อาจยังใช้ index บ้างในบางกรณี แต่ไม่ efficient)

-- หลัก "Leftmost Prefix Rule":
-- Multi-column index (a, b, c) สามารถใช้ได้กับ:
-- - WHERE a = ?
-- - WHERE a = ? AND b = ?
-- - WHERE a = ? AND b = ? AND c = ?
-- - ORDER BY a, b, c
-- ไม่สามารถใช้ได้กับ:
-- - WHERE b = ? (ข้าม a)
-- - WHERE c = ? (ข้าม a, b)

-- กฎการเลือกลำดับ columns:
-- 1. Equality conditions ก่อน (=)
-- 2. Range conditions หลัง (>, <, BETWEEN)
-- 3. High cardinality ก่อน low cardinality (ถ้าทุกอย่างเท่ากัน)

-- ตัวอย่าง: Optimal column order
-- Query: WHERE status = 'completed' AND price BETWEEN 1000 AND 5000

-- ดี: equality (status) ก่อน range (price)
CREATE INDEX idx_good_order ON shop.orders(status, total_amount);

-- ไม่ดี: range ก่อน equality
CREATE INDEX idx_bad_order ON shop.orders(total_amount, status);

EXPLAIN SELECT * FROM shop.orders 
WHERE status = 'completed' AND total_amount BETWEEN 1000 AND 5000;
-- idx_good_order จะ efficient กว่า

-- Multi-column index vs Multiple single-column indexes
-- Multi-column index ดีกว่าเมื่อ WHERE clause มักใช้ทั้งสอง columns พร้อมกัน
-- Single-column indexes ดีกว่าเมื่อแต่ละ column ถูก query แยกกันด้วย

-- ตรวจสอบ index ที่ overlap หรือซ้ำซ้อน
SELECT 
    indexname,
    indexdef
FROM pg_indexes
WHERE tablename = 'orders'
ORDER BY indexname;
```

## 10. Index-Only Scan

```sql
-- Index-Only Scan: Query ที่ตอบได้จาก index โดยไม่ต้องเข้า heap table
-- เกิดขึ้นเมื่อ: ทุก column ที่ query ต้องการมีอยู่ใน index

-- Setup
CREATE TABLE index_scan_demo (
    id      SERIAL PRIMARY KEY,
    email   VARCHAR(255),
    name    VARCHAR(100),
    status  VARCHAR(20)
);

INSERT INTO index_scan_demo (email, name, status)
SELECT 
    'user' || i || '@example.com',
    'User ' || i,
    CASE i % 3 WHEN 0 THEN 'active' WHEN 1 THEN 'inactive' ELSE 'pending' END
FROM generate_series(1, 100000) AS i;

-- สร้าง covering index
CREATE INDEX idx_covering_demo ON index_scan_demo(status) INCLUDE (email, name);

-- Query นี้ทำ Index-Only Scan:
EXPLAIN (ANALYZE, BUFFERS) 
SELECT email, name FROM index_scan_demo WHERE status = 'active';
-- Output: Index Only Scan ... Heap Fetches: 0

-- Query นี้ทำ Index Scan (ต้องการ id ซึ่งไม่มีใน index):
EXPLAIN (ANALYZE, BUFFERS)
SELECT id, email, name FROM index_scan_demo WHERE status = 'active';
-- Output: Index Scan ... Heap Fetches: N

-- VACUUM จำเป็นสำหรับ Index-Only Scan!
-- Index-Only Scan ใช้ visibility map
-- ต้องทำ VACUUM ให้ pages เป็น "all visible"
VACUUM ANALYZE index_scan_demo;

-- หลัง VACUUM:
EXPLAIN (ANALYZE, BUFFERS)
SELECT email, name FROM index_scan_demo WHERE status = 'active';
-- Heap Fetches จะลดลง
```

## 11. Bitmap Index Scan

```sql
-- Bitmap Index Scan: PostgreSQL ใช้เมื่อ query return rows จำนวนมาก
-- กระบวนการ:
-- 1. สร้าง Bitmap ของ pages/rows ที่ตรงกับ index conditions
-- 2. Fetch pages ใน physical order
-- 3. Filter rows จาก heap

-- เกิดขึ้นเมื่อ: query return ~1-20% ของ table (ระหว่าง Index Scan และ Seq Scan)

-- ตัวอย่าง: Low selectivity query
EXPLAIN ANALYZE 
SELECT * FROM shop.orders WHERE status = 'completed';
-- อาจเห็น: Bitmap Heap Scan on orders
--           -> Bitmap Index Scan on idx_orders_status

-- Bitmap AND / OR (combining multiple indexes)
CREATE INDEX idx_orders_status2 ON shop.orders(status);
CREATE INDEX idx_orders_payment ON shop.orders(payment_method);

EXPLAIN ANALYZE
SELECT * FROM shop.orders 
WHERE status = 'completed' AND payment_method = 'credit_card';
-- PostgreSQL อาจใช้ BitmapAnd (scan two indexes + AND the bitmaps)

EXPLAIN ANALYZE
SELECT * FROM shop.orders 
WHERE status = 'completed' OR payment_method = 'credit_card';
-- PostgreSQL อาจใช้ BitmapOr (scan two indexes + OR the bitmaps)

-- เมื่อไหร่ PostgreSQL เลือก scan type:
-- Seq Scan:          ไม่มี index, หรือ return rows > ~20%
-- Index Scan:        index ดี, return rows น้อย (<1-2%)
-- Index Only Scan:   covering index, ทุก column อยู่ใน index
-- Bitmap Index Scan: return rows ปานกลาง (1-20%)
```

### 12.3 Red Flags ที่ต้องสังเกต

```sql
-- ❌ Red Flag 1: Seq Scan บน table ใหญ่พร้อม filter
-- ควรเป็น Index Scan แต่ไม่มี index
-- Solution: สร้าง index
EXPLAIN SELECT * FROM index_demo.orders_large WHERE status = 'completed';
-- Output: Seq Scan (ถ้าไม่มี index)

-- ❌ Red Flag 2: Hash Join บน table ใหญ่ (memory issue)
-- spill to disk เกิดขึ้นเมื่อ hash ใหญ่เกิน work_mem
-- Solution: เพิ่ม work_mem หรือ สร้าง index

-- ❌ Red Flag 3: Nested Loop บน table ใหญ่
-- O(N*M) complexity — ช้ามากถ้า outer table ใหญ่
-- Solution: สร้าง index บน inner table's join column

EXPLAIN ANALYZE SELECT * FROM shop.orders WHERE status = 'pending';

-- Solution: ANALYZE table_name;

-- ❌ Red Flag 5: Materialize node
-- PostgreSQL ต้อง buffer result สำหรับ nested loops
-- อาจหมายถึง missing index

-- PgHero: https://github.com/ankane/pghero
-- pgBadger: log analyzer
-- EXPLAIN.depesz.com: paste explain output
```

### 12.4 ตัวอย่าง: Identify และ Fix Slow Query

```sql
-- Step 1: ระบุ slow query
SELECT 
    pid,
    now() - query_start AS duration,
    query
FROM pg_stat_activity
WHERE state = 'active'
  AND now() - query_start > INTERVAL '1 second'
ORDER BY duration DESC;

-- หรือดูจาก pg_stat_statements (ต้องเปิด extension)
CREATE EXTENSION IF NOT EXISTS pg_stat_statements;

SELECT 
    query,
    calls,
    total_exec_time / calls AS avg_time_ms,
    rows / calls AS avg_rows,
    total_exec_time
FROM pg_stat_statements
ORDER BY total_exec_time DESC
LIMIT 10;

-- Step 2: รัน EXPLAIN ANALYZE
EXPLAIN (ANALYZE, BUFFERS)
SELECT 
    c.first_name, c.last_name,
    SUM(o.total_amount) AS total_spent
FROM shop.customers c
JOIN shop.orders o ON c.customer_id = o.customer_id
WHERE o.status = 'completed'
  AND o.created_at > '2024-01-01'
GROUP BY c.customer_id, c.first_name, c.last_name
ORDER BY total_spent DESC
LIMIT 10;

-- Step 3: สังเกต
-- - Seq Scan บน orders table

-- - high buffer read

CREATE INDEX idx_optimize_orders ON shop.orders(status, created_at) 
    INCLUDE (customer_id, total_amount);

-- Step 5: รัน EXPLAIN อีกครั้ง
EXPLAIN (ANALYZE, BUFFERS)
SELECT ...;  -- same query
-- ควรเห็น Index Scan แทน Seq Scan
-- Execution time ควรลดลงมาก
```

## 13. Index Bloat และ REINDEX

### 13.1 Index Bloat คืออะไร?

```sql
-- Index Bloat เกิดขึ้นเมื่อ:
-- 1. UPDATE rows หลายครั้ง → dead tuples ใน index
-- 2. DELETE rows จำนวนมาก
-- 3. Autovacuum ทำงานไม่ทัน

-- ตรวจสอบ Index Bloat
SELECT 
    schemaname,
    tablename,
    indexname,
    pg_size_pretty(pg_relation_size(indexrelid)) AS index_size,
    idx_scan,
    idx_tup_read,
    idx_tup_fetch
FROM pg_stat_user_indexes
ORDER BY pg_relation_size(indexrelid) DESC
LIMIT 20;

-- ดูรายละเอียด bloat (ต้องการ extension pgstattuple)
CREATE EXTENSION IF NOT EXISTS pgstattuple;

SELECT * FROM pgstattuple('idx_orders_status');
-- เช็ค dead_tuple_percent

-- Bloat estimation query (ไม่ต้องใช้ extension)
SELECT 
    i.relname AS index_name,
    pg_size_pretty(pg_relation_size(i.oid)) AS index_size,
    t.relname AS table_name,
    pg_size_pretty(pg_relation_size(t.oid)) AS table_size,
    ROUND(100 * pg_relation_size(i.oid) / NULLIF(pg_relation_size(t.oid), 0), 2) AS ratio
FROM pg_class i
JOIN pg_index ix ON i.oid = ix.indexrelid
JOIN pg_class t ON ix.indrelid = t.oid
WHERE t.relkind = 'r'
  AND i.relkind = 'i'
  AND t.relname NOT LIKE 'pg_%'
ORDER BY pg_relation_size(i.oid) DESC;
```

### 13.2 REINDEX

```sql
-- REINDEX: สร้าง index ใหม่ (ลบ bloat)
-- ข้อเสีย: Lock table ระหว่างทำ!

-- REINDEX index เดียว (ต้อง lock!)
REINDEX INDEX idx_orders_status;

-- REINDEX ทุก index ของ table (ต้อง lock!)
REINDEX TABLE shop.orders;

-- REINDEX ทุก index ของ database (ต้อง lock!)
REINDEX DATABASE mydb;

-- REINDEX CONCURRENTLY (PostgreSQL 12+): ไม่ต้อง lock!
REINDEX INDEX CONCURRENTLY idx_orders_status;
REINDEX TABLE CONCURRENTLY shop.orders;

-- หรือ: สร้าง index ใหม่แล้วลบเก่า (manual concurrent reindex)
-- Step 1: สร้าง index ใหม่แบบ concurrent
CREATE INDEX CONCURRENTLY idx_orders_status_new ON shop.orders(status);

-- Step 2: Drop index เก่า
DROP INDEX idx_orders_status;

-- Step 3: Rename index ใหม่
ALTER INDEX idx_orders_status_new RENAME TO idx_orders_status;
```

### 13.3 Autovacuum Tuning เพื่อลด Bloat

```sql
-- ดู autovacuum settings
SHOW autovacuum;
SHOW autovacuum_vacuum_scale_factor;     -- default: 0.2 (20%)
SHOW autovacuum_analyze_scale_factor;    -- default: 0.1 (10%)

-- ปรับสำหรับ table ที่ update บ่อย
ALTER TABLE shop.orders SET (
    autovacuum_vacuum_scale_factor = 0.01,    -- 1% (เร็วกว่า default)
    autovacuum_analyze_scale_factor = 0.005,  -- 0.5%
    autovacuum_vacuum_cost_delay = 2          -- ms
);

-- Manual VACUUM
VACUUM shop.orders;           -- vacuum ธรรมดา (ไม่ lock, คืน dead tuples)
VACUUM ANALYZE shop.orders;   -- vacuum + update statistics
VACUUM FULL shop.orders;      -- lock table, rebuild table, reclaim disk space
VACUUM FREEZE shop.orders;    -- ป้องกัน transaction ID wraparound
```

## 14. pg_stat_user_indexes

```sql
-- pg_stat_user_indexes: statistics สำหรับ indexes ทั้งหมด

SELECT * FROM pg_stat_user_indexes LIMIT 5;

-- Columns สำคัญ:
-- indexrelid: index OID
-- relid: table OID
-- schemaname: schema name
-- relname: table name
-- indexrelname: index name
-- idx_scan: จำนวนครั้งที่ index scan ถูกใช้
-- idx_tup_read: จำนวน index entries ที่อ่าน
-- idx_tup_fetch: จำนวน rows ที่ fetch จาก heap

-- ค้นหา indexes ที่ใช้บ่อยสุด
SELECT 
    schemaname || '.' || relname AS table_name,
    indexrelname AS index_name,
    idx_scan AS total_scans,
    pg_size_pretty(pg_relation_size(indexrelid)) AS index_size,
    idx_tup_read,
    idx_tup_fetch
FROM pg_stat_user_indexes
WHERE schemaname NOT IN ('pg_catalog', 'pg_toast')
ORDER BY idx_scan DESC
LIMIT 20;

-- ค้นหา indexes ที่ไม่ค่อยได้ใช้ (candidates for removal)
SELECT 
    schemaname || '.' || relname AS table_name,
    indexrelname AS index_name,
    idx_scan,
    pg_size_pretty(pg_relation_size(indexrelid)) AS wasted_space
FROM pg_stat_user_indexes
WHERE idx_scan < 10  -- ใช้น้อยกว่า 10 ครั้ง
  AND indexrelid NOT IN (
      SELECT conindid FROM pg_constraint WHERE contype IN ('p', 'u', 'x')
  )
ORDER BY pg_relation_size(indexrelid) DESC;

-- Reset statistics (เริ่มนับใหม่)
SELECT pg_stat_reset();

-- Monitoring dashboard query
SELECT 
    i.relname AS table_name,
    ix.relname AS index_name,
    pg_size_pretty(pg_relation_size(ix.oid)) AS index_size,
    idx.idx_scan,
    idx.idx_tup_read,
    idx.idx_tup_fetch,
    CASE 
        WHEN idx.idx_scan = 0 THEN 'UNUSED'
        WHEN idx.idx_scan < 100 THEN 'RARELY USED'
        WHEN idx.idx_scan < 1000 THEN 'OCCASIONALLY USED'
        ELSE 'FREQUENTLY USED'
    END AS usage_level
FROM pg_class i
JOIN pg_index pig ON i.oid = pig.indrelid
JOIN pg_class ix ON pig.indexrelid = ix.oid
JOIN pg_stat_user_indexes idx ON ix.oid = idx.indexrelid
WHERE i.relkind = 'r'
  AND NOT pig.indisprimary
  AND NOT pig.indisunique
ORDER BY idx.idx_scan ASC, pg_relation_size(ix.oid) DESC;
```

```sql
-- Query ที่ช้า:

EXPLAIN ANALYZE

CREATE INDEX idx_ex2_orders_customer ON workshop.large_orders(customer_id, status)

    DATE_TRUNC('day', created_at) AS day,

-- ===========================================
-- Final: Index Maintenance Script
-- ===========================================
DO $$
DECLARE
    v_index TEXT;
BEGIN
    -- Find bloated indexes (size > 10MB, idx_scan < 10)
    FOR v_index IN
        SELECT indexrelname
        FROM pg_stat_user_indexes
        WHERE pg_relation_size(indexrelid) > 10485760  -- 10MB
          AND idx_scan < 10
          AND indexrelid NOT IN (
              SELECT conindid FROM pg_constraint WHERE contype IN ('p', 'u')
          )
    LOOP
        RAISE NOTICE 'Consider dropping unused index: %', v_index;
    END LOOP;
    
    -- Suggest REINDEX for bloated tables
    FOR v_index IN
        SELECT i.relname
        FROM pg_class i
        JOIN pg_stat_user_tables t ON i.relname = t.relname
        WHERE t.n_dead_tup > t.n_live_tup * 0.1  -- > 10% dead tuples
          AND i.relkind = 'i'
        LIMIT 10
    LOOP
        RAISE NOTICE 'Consider REINDEX: %', v_index;
    END LOOP;
END;
$$;
```

## สรุปบทที่ 5

### Cheat Sheet: Index Decision Tree

```
ต้องการ optimize query ไหม?
│
├── มี Index อยู่แล้ว?
│   ├── ใช่ → EXPLAIN ANALYZE → ดูว่า Index ถูกใช้ไหม
│   │         ├── ถูกใช้แล้ว → ปัญหาอยู่ที่อื่น (statistics? work_mem?)
│   │         └── ไม่ถูกใช้ → เช็ค predicate match
│   │
│   └── ไม่มี → สร้าง Index
│               ├── WHERE = value → B-Tree
│               ├── WHERE LIKE 'prefix%' → B-Tree หรือ SP-GiST
│               ├── WHERE column @> array → GIN
│               ├── WHERE jsonb @> json → GIN
│               ├── Full-text search → GIN หรือ GiST
│               ├── Range query บน timestamp → B-Tree หรือ BRIN
│               ├── Geometric/IP → GiST หรือ SP-GiST
│               └── Equality only → B-Tree (Hash ก็ได้)
│
├── Query ยังช้า?
│   ├── Low cardinality? → Partial Index
│   ├── Function ใน WHERE? → Expression Index
│   ├── SELECT หลาย columns? → Covering Index (INCLUDE)
│   └── Multi-column WHERE? → Multi-column Index (equality ก่อน)
│
└── Index ใหญ่เกิน หรือ write ช้า?
    ├── Dead tuples → VACUUM ANALYZE
    ├── Fragmentation → REINDEX CONCURRENTLY
    └── Unused indexes → DROP INDEX
```

| เรื่อง | คำสั่งสำคัญ |
|-------|------------|
| สร้าง Index | CREATE INDEX, CREATE UNIQUE INDEX |
| ลบ Index | DROP INDEX |
| Rebuild | REINDEX, REINDEX CONCURRENTLY |
| ดู plan | EXPLAIN, EXPLAIN ANALYZE |
| Monitor | pg_stat_user_indexes |
| Maintenance | VACUUM ANALYZE, ANALYZE |
